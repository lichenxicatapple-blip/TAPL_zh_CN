% Pygments 的 friendly 样式默认把注释设为斜体。英文注释尚可接受，但这会
% 令中文和标点切换到不协调的斜体字形。保留 friendly 的注释颜色，只取消
% Comment、Comment.Hashbang、Comment.Multiline、Comment.PreprocFile 和
% Comment.Single 的斜体。minted 每次排版代码前都会重新载入样式，因此把
% 覆盖逻辑附加到其样式初始化末尾。
\makeatletter
\newcommand{\taplmintedcommentupright}{%
  \@namedef{PYG@tok@c}{\def\PYG@tc####1{\textcolor[rgb]{0.38,0.63,0.69}{####1}}}%
  \@namedef{PYG@tok@ch}{\def\PYG@tc####1{\textcolor[rgb]{0.38,0.63,0.69}{####1}}}%
  \@namedef{PYG@tok@cm}{\def\PYG@tc####1{\textcolor[rgb]{0.38,0.63,0.69}{####1}}}%
  \@namedef{PYG@tok@cpf}{\def\PYG@tc####1{\textcolor[rgb]{0.38,0.63,0.69}{####1}}}%
  \@namedef{PYG@tok@c1}{\def\PYG@tc####1{\textcolor[rgb]{0.38,0.63,0.69}{####1}}}%
}
\apptocmd{\minted@defstyle}{\taplmintedcommentupright}{}{%
  \PackageError{tapl}{Unable to normalize minted comment style}{}%
}
\makeatother
